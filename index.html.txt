(function () {
  const root = document.getElementById('csr-photo');
  if (!root || root.dataset.initialized) return;
  root.dataset.initialized = 'true';

  // Replace with your WhatsApp number: country code + number, digits only.
  const WHATSAPP_NUMBER = '919000000000';
  const whatsappURL = 'https://wa.me/' + WHATSAPP_NUMBER;

  const waLink = root.querySelector('#csr-whatsapp');
  waLink.href = whatsappURL;
  root.querySelector('#csr-year').textContent = new Date().getFullYear();

  const menu = root.querySelector('.csr-menu');
  const nav = root.querySelector('.csr-nav');
  menu.addEventListener('click', function () {
    nav.classList.toggle('csr-open');
  });
  nav.querySelectorAll('a').forEach(function (link) {
    link.addEventListener('click', function () {
      nav.classList.remove('csr-open');
    });
  });

  root.querySelectorAll('.csr-filters button').forEach(function (button) {
    button.addEventListener('click', function () {
      root.querySelectorAll('.csr-filters button').forEach(function (b) {
        b.classList.remove('active');
      });
      button.classList.add('active');
      const filter = button.dataset.filter;
      root.querySelectorAll('.csr-gallery-grid img').forEach(function (img) {
        img.classList.toggle('csr-hidden',
          filter !== 'all' && img.dataset.category !== filter);
      });
    });
  });

  // Simple image lightbox
  root.querySelectorAll('.csr-gallery-grid img').forEach(function (img) {
    img.addEventListener('click', function () {
      const overlay = document.createElement('div');
      overlay.className = 'csr-lightbox';
      const largeImage = document.createElement('img');
      largeImage.src = img.src;
      largeImage.alt = img.alt;
      const close = document.createElement('button');
      close.type = 'button';
      close.textContent = '×';
      close.setAttribute('aria-label', 'Close image');
      overlay.append(largeImage, close);
      root.appendChild(overlay);
      function closeOverlay() { overlay.remove(); }
      close.addEventListener('click', closeOverlay);
      overlay.addEventListener('click', function (e) {
        if (e.target === overlay) closeOverlay();
      });
      document.addEventListener('keydown', function onKey(e) {
        if (e.key === 'Escape') {
          closeOverlay();
          document.removeEventListener('keydown', onKey);
        }
      });
    });
  });

  // Replace this sample text with real client testimonials.
  const reviews = [
    { text: 'Beautiful memories deserve beautiful photographs.', name: 'CSR Photography' },
    { text: 'Every special moment deserves to be remembered.', name: 'Our Photography Promise' },
    { text: 'Your story, captured with care and creativity.', name: 'CSR Photography' }
  ];
  let reviewIndex = 0;
  function showReview() {
    root.querySelector('#csr-review-text').textContent = '“' + reviews[reviewIndex].text + '”';
    root.querySelector('#csr-review-name').textContent = reviews[reviewIndex].name;
  }
  root.querySelector('#csr-prev').addEventListener('click', function () {
    reviewIndex = (reviewIndex + reviews.length - 1) % reviews.length;
    showReview();
  });
  root.querySelector('#csr-next').addEventListener('click', function () {
    reviewIndex = (reviewIndex + 1) % reviews.length;
    showReview();
  });

  root.querySelector('#csr-booking').addEventListener('submit', function (e) {
    e.preventDefault();
    const name = root.querySelector('#csr-name').value.trim();
    const phone = root.querySelector('#csr-phone').value.trim();
    const service = root.querySelector('#csr-type').value;
    const date = root.querySelector('#csr-date').value || 'Not specified';
    const message = root.querySelector('#csr-message').value.trim() || 'No additional details';
    const text = [
      'Hello CSR Photography WGL!',
      'I would like to enquire about a photography booking.',
      'Name: ' + name,
      'Phone: ' + phone,
      'Service: ' + service,
      'Event date: ' + date,
      'Details: ' + message
    ].join('\n');
    window.open(whatsappURL + '?text=' + encodeURIComponent(73866992244), '_blank', 'noopener');
  });
})();
